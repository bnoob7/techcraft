--
-- 1. Create the 'owners' table
-- Used by: ownerController.js
-- The owner_id will serve as the Foreign Key for products and orders.
--
CREATE TABLE owners (
    owner_id INT AUTO_INCREMENT PRIMARY KEY,
    owner_name VARCHAR(255) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE, -- Email is used for login and must be unique
    password VARCHAR(255) NOT NULL, -- Stores the hashed password
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--
-- 2. Create the 'customers' table
-- Used by: customerController.js and orderController.js
-- The 'id' column is the customer's unique identifier.
--
CREATE TABLE customers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE, -- Email is used for login and must be unique
    password VARCHAR(255) NOT NULL, -- Stores the hashed password
    contact_number VARCHAR(20) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--
-- 3. Create the 'admin' table
-- Used by: adminController.js
-- This is a separate table for administrator accounts.
--
CREATE TABLE admin (
    admin_id INT AUTO_INCREMENT PRIMARY KEY, -- Used to sign the JWT token
    name VARCHAR(255) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--
-- 4. Create the 'products' table
-- Used by: productController.js and cartController.js (for joins)
-- owner_id is a Foreign Key referencing the 'owners' table.
--
CREATE TABLE products (
    product_id INT AUTO_INCREMENT PRIMARY KEY,
    product_name VARCHAR(255) NOT NULL,
    description TEXT,
    price DECIMAL(10, 2) NOT NULL,
    photo VARCHAR(255),
    owner_id INT NOT NULL,
    shop_name VARCHAR(255) NOT NULL,
    category VARCHAR(100) NOT NULL, -- New required column
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    -- Foreign Key Constraint
    FOREIGN KEY (owner_id) REFERENCES owners(owner_id) 
    -- Assuming a one-to-many relationship: One owner can have many products.
    -- Consider adding ON DELETE CASCADE or RESTRICT as needed for data integrity.
);

--
-- 5. Create the 'cart' table
-- Used by: cartController.js
-- This acts as a junction table between customers and products.
-- A composite Primary Key is typically used for a cart to prevent duplicate entries for the same product by the same user.
--
CREATE TABLE cart (
    user_id INT NOT NULL, -- References customers.id (customer is the user)
    product_id INT NOT NULL, -- References products.product_id
    quantity INT NOT NULL,
    added_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    PRIMARY KEY (user_id, product_id), -- Composite primary key to ensure uniqueness
    
    -- Foreign Key Constraints
    FOREIGN KEY (user_id) REFERENCES customers(id),
    FOREIGN KEY (product_id) REFERENCES products(product_id),
    
    -- Used for the ON DUPLICATE KEY UPDATE logic in cartController.js
    UNIQUE KEY unique_cart_item (user_id, product_id) 
);

--
-- 6. Create the 'orders' table
-- Used by: orderController.js
-- References both the customer and the owner who fulfills the order.
--
CREATE TABLE orders (
    order_id INT AUTO_INCREMENT PRIMARY KEY,
    customer_id INT NOT NULL, -- References customers.id
    owner_id INT NOT NULL, -- References owners.owner_id (the owner receiving the order)
    shop_name VARCHAR(255) NOT NULL,
    items JSON NOT NULL, -- Stored as a JSON string (used JSON.stringify in controller)
    total_amount DECIMAL(10, 2) NOT NULL,
    order_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    -- Foreign Key Constraints
    FOREIGN KEY (customer_id) REFERENCES customers(id),
    FOREIGN KEY (owner_id) REFERENCES owners(owner_id)
);
